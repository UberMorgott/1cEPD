# Интерфейс на Vue — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Рабочий интерфейс, где видно найденные поломки идентификаторов, остатки квот и кому выставлять счёт, с живым обновлением и без единой кнопки «обновить» как единственного способа получить свежие данные.

**Architecture:** Vue 3 + Vite + TypeScript + PrimeVue. Данные берутся только через свой бэкенд. Поток изменений приходит по SSE в один Pinia store, экраны на него подписаны. Собранный фронт вшивается в бинарник через `embed` и отдаётся тем же сервером.

**Tech Stack:** Vue 3, Vite, TypeScript, Vue Router, Pinia, PrimeVue.

**Предыдущие планы:** `2026-09-02-foundation.md`, `2026-09-02-identifiers-and-billing.md`

## Что даёт бэкенд

| Метод | Ответ |
|---|---|
| `POST /api/login` | `{"ok":true}` плюс cookie сессии |
| `POST /api/logout` | `{"ok":true}` |
| `GET /api/health` | `{"status":"ok"}` |
| `GET /api/events` | поток SSE, события `snapshot.completed`, keep-alive комментарием |
| `GET /api/identifiers` | `{"identifiers":[{edoId,clientName,inn,kpp,login,owner,ownerCode,tariffs,limit,packets,firstSeen,lastSeen,period}]}` |
| `GET /api/anomalies?all=1` | `{"anomalies":[{id,edoId,kind,confidence,fingerprint,inn,kpp,clientName,login,details,detectedAt,acknowledged}]}` |
| `POST /api/anomalies/ack` | тело `{edoId,fingerprint,reason}` |
| `GET /api/billing?period=2026-08` | `{period,takenAt,clients:[…],needsAction:[…],totalDue}` |

Ошибки приходят как `{"error":{"code":…,"message":…}}`, неавторизованный запрос — `401`.

Виды находок: `orphan_with_traffic` (высокая уверенность, трафик мимо тарифа),
`owner_lost` (высокая, владелец пропал), `orphan_idle` (низкая, наблюдение).

```ts
/** Ошибка, пришедшая от нашего бэкенда в едином формате. */
export class ApiError extends Error {
  constructor(
    readonly status: number,
    readonly code: string,
    message: string,
  ) {
    super(message)
  }
}

interface ErrorBody {
  error?: { code?: string; message?: string }
}

/**
 * request выполняет запрос к нашему API.
 *
 * credentials: 'same-origin' обязателен: сессия живёт в cookie, без этого
 * браузер её не пошлёт и каждый запрос будет получать 401.
 */
async function request<T>(path: string, init: RequestInit = {}): Promise<T> {
  const response = await fetch(path, {
    ...init,
    credentials: 'same-origin',
    headers: {
      'Content-Type': 'application/json',
      ...(init.headers ?? {}),
    },
  })

  if (response.status === 204) {
    return undefined as T
  }

  const text = await response.text()
  const payload = text ? (JSON.parse(text) as unknown) : null

  if (!response.ok) {
    const body = payload as ErrorBody | null
    throw new ApiError(
      response.status,
      body?.error?.code ?? 'unknown',
      body?.error?.message ?? `Запрос завершился ошибкой ${response.status}`,
    )
  }

  return payload as T
}

export const api = {
  login: (login: string, password: string) =>
    request<{ ok: boolean }>('/api/login', {
      method: 'POST',
      body: JSON.stringify({ login, password }),
    }),

  logout: () => request<{ ok: boolean }>('/api/logout', { method: 'POST' }),

  anomalies: (includeAcknowledged = false) =>
    request<{ anomalies: Anomaly[] }>(
      `/api/anomalies${includeAcknowledged ? '?all=1' : ''}`,
    ),

  acknowledge: (edoId: string, fingerprint: string, reason: string) =>
    request<{ ok: boolean }>('/api/anomalies/ack', {
      method: 'POST',
      body: JSON.stringify({ edoId, fingerprint, reason }),
    }),

  identifiers: () => request<{ identifiers: Identifier[] }>('/api/identifiers'),

  billing: (period?: string) =>
    request<BillingReport>(`/api/billing${period ? `?period=${period}` : ''}`),
}

export interface Anomaly {
  id: number
  edoId: string
  kind: 'orphan_with_traffic' | 'owner_lost' | 'orphan_idle'
  confidence: 'high' | 'low'
  fingerprint: string
  inn: string
  kpp: string
  clientName: string
  login: string
  details: string
  detectedAt: string
  acknowledged: boolean
}

export interface Identifier {
  edoId: string
  clientName: string
  inn: string
  kpp: string
  login: string
  owner: string
  ownerCode: string
  tariffs: string
  limit: number | null
  packets: number
  firstSeen: string
  lastSeen: string
  period: string
}

export interface BillingClient {
  edoId: string
  clientName: string
  inn: string
  kpp: string
  login: string
  owner: string
  tariffs: string
  limit: number | null
  used: number
  remaining: number | null
  billable: number
  amount: string
  overLimit: boolean
  lowRemainder: boolean
}

export interface BillingReport {
  period: string
  takenAt?: string
  clients: BillingClient[]
  needsAction: BillingClient[]
  totalDue: string
  empty?: boolean
}
```

```ts
import { defineStore } from 'pinia'
import { ref } from 'vue'
import { api, ApiError } from '../api/client'

export const useSessionStore = defineStore('session', () => {
  const authenticated = ref(false)
  const error = ref('')
  const busy = ref(false)

  async function login(name: string, password: string): Promise<boolean> {
    busy.value = true
    error.value = ''
    try {
      await api.login(name, password)
      authenticated.value = true
      return true
    } catch (err) {
      error.value = err instanceof ApiError ? err.message : 'Не удалось войти.'
      return false
    } finally {
      busy.value = false
    }
  }

  async function logout() {
    try {
      await api.logout()
    } finally {
      authenticated.value = false
    }
  }

  /** Помечает сессию потерянной. Вызывается из клиента API при 401. */
  function expire() {
    authenticated.value = false
  }

  return { authenticated, error, busy, login, logout, expire }
})
```

- [ ] **Step 2: Реализовать `live.ts`**

```ts
import { defineStore } from 'pinia'
import { ref } from 'vue'

/** Событие из потока сервера. */
export interface LiveEvent {
  kind: string
  payload: Record<string, unknown>
}

export const useLiveStore = defineStore('live', () => {
  const connected = ref(false)
  const lastEvent = ref<LiveEvent | null>(null)
  const updatedAt = ref<Date | null>(null)

  let source: EventSource | null = null
  let retryDelay = 1000

  function handle(kind: string, event: MessageEvent) {
    let payload: Record<string, unknown> = {}
    try {
      payload = JSON.parse(event.data) as Record<string, unknown>
    } catch {
      // Событие без разбираемых данных — не повод падать, важен сам факт.
    }
    lastEvent.value = { kind, payload }
    updatedAt.value = new Date()
  }

  function connect() {
    if (source) return

    source = new EventSource('/api/events', { withCredentials: true })

    source.onopen = () => {
      connected.value = true
      retryDelay = 1000
    }

    for (const kind of ['snapshot.completed', 'identifier.alert', 'programs.synced']) {
      source.addEventListener(kind, (event) => handle(kind, event as MessageEvent))
    }

    source.onerror = () => {
      connected.value = false
      source?.close()
      source = null
      // Переподключаемся с нарастающей паузой: сервер мог перезапуститься,
      // а долбить его каждую секунду незачем.
      setTimeout(connect, retryDelay)
      retryDelay = Math.min(retryDelay * 2, 30000)
    }
  }

  function disconnect() {
    source?.close()
    source = null
    connected.value = false
  }

  return { connected, lastEvent, updatedAt, connect, disconnect }
})
```

```go
// Package web отдаёт собранный интерфейс, вшитый в бинарник.
package web

import (
	"embed"
	"fmt"
	"io/fs"
	"net/http"
	"path"
	"strings"
)

//go:embed all:dist
var assets embed.FS

// Handler отдаёт файлы собранного фронта.
//
// Неизвестные пути возвращают index.html: маршруты обрабатывает Vue Router
// на клиенте, и без этого перезагрузка страницы по адресу вроде /billing дала бы 404.
func Handler() (http.Handler, error) {
	root, err := fs.Sub(assets, "dist")
	if err != nil {
		return nil, fmt.Errorf("web: не найден каталог сборки: %w", err)
	}

	files := http.FileServer(http.FS(root))

	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		name := strings.TrimPrefix(path.Clean(r.URL.Path), "/")
		if name == "" {
			name = "index.html"
		}

		if _, err := fs.Stat(root, name); err != nil {
			r = r.Clone(r.Context())
			r.URL.Path = "/"
		}
		files.ServeHTTP(w, r)
	}), nil
}
```
